// api/types/draft.ts

export interface DraftLead {
  id: string;
  estado: 'draft' | 'activo' | 'calificado' | 'oportunidad' | 'ganado' | 'perdido';
  datos: LeadData;
  completitud: number; // 0-100%
  validaciones: ValidationStatus;
  createdAt: string;
  updatedAt: string;
  createdBy: string; // ID del usuario que creó
  notes?: string;
}

export interface LeadData {
  // Campos obligatorios
  nombre: string;
  email: string;
  telefono?: string;
  empresa: string;

  // Campos opcionales
  cargo?: string;
  industria?: string;
  pais?: string;
  presupuesto?: number;
  timeline?: string; // Ej: '1-3 meses', '3-6 meses'

  fuente?: string; // LinkedIn, Referencia, Evento, etc.
  etapaVenta?: string;
  valorPotencial?: number;
  notas?: string;
}

export interface ValidationStatus {
  email: { valid: boolean; message?: string };
  telefono: { valid: boolean; message?: string };
  empresa: { valid: boolean; message?: string };
  duplicado: { isDuplicate: boolean; duplicateId?: string };
  completitud: {
    required: number; // Campos obligatorios completados
    total: number;    // Total de campos
    percentage: number;
  };
}

export interface CreateDraftLeadDTO {
  nombre: string;
  email: string;
  empresa: string;

  // ... otros campos opcionales
}

export interface UpdateDraftLeadDTO {
  nombre?: string;
  email?: string;
  telefono?: string;
  empresa?: string;
  cargo?: string;
  // ... todos los campos son opcionales para edición parcial

## 💻 Componentes React - Integración

### 1. DraftLeadForm (Crear/Editar)
```tsx
// components/Drafts/DraftLeadForm.tsx

interface DraftLeadFormProps {
  draftId?: string;
  onSuccess: (draft: DraftLead) => void;
  onError: (error: ApiError) => void;
}

export const DraftLeadForm: React.FC<DraftLeadFormProps> = ({
  draftId,
  onSuccess,
  onError
}) => {
  const [formData, setFormData] = useState<CreateDraftLeadDTO>({});
  const [validations, setValidations] = useState<ValidationStatus>({});
  const [completitud, setCompletitud] = useState(0);
  const [loading, setLoading] = useState(false);

  // Hook para gestionar el draft
  const { saveDraft, updateDraft } = useDraft();

  const handleFieldChange = async (field: string, value: string) => {
    const newData = { ...formData, [field]: value };
    setFormData(newData);

    // Validar en tiempo real
    if (field === 'email') {
      const emailValidation = await validateEmail(value);
      setValidations(prev => ({
        ...prev,
        email: emailValidation
      }));
    }

    // Actualizar completitud
    const completeness = calculateCompleteness(newData);
    setCompletitud(completeness);
  };

  const handleSave = async () => {
    setLoading(true);
    try {
      const draft = draftId
        ? await updateDraft(draftId, formData)
        : await saveDraft(formData);

      onSuccess(draft);
    } catch (error) {
      onError(error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <form>
      {/* Campos del formulario */}
      <input
        placeholder="Nombre *"
        value={formData.nombre}
        onChange={(e) => handleFieldChange('nombre', e.target.value)}
        required
      />

      <input
        placeholder="Email *"
        type="email"
        value={formData.email}
        onChange={(e) => handleFieldChange('email', e.target.value)}
        required
      />
      {validations.email && !validations.email.valid && (
        <span className="error">{validations.email.message}</span>
      )}

      {/* Mostrar progreso de completitud */}
      <ProgressBar value={completitud} />

      <button onClick={handleSave} disabled={loading}>
        {loading ? 'Guardando...' : 'Guardar Draft'}
      </button>
    </form>
  );
};
```

### 3. DraftConfirmModal
```tsx
// components/Drafts/DraftConfirmModal.tsx

interface DraftConfirmModalProps {
  draft: DraftLead;
  isOpen: boolean;
  onConfirm: (data: ConfirmDraftDTO) => void;
  onCancel: () => void;
}

export const DraftConfirmModal: React.FC<DraftConfirmModalProps> = ({
  draft,
  isOpen,
  onConfirm,
  onCancel
}) => {
  const [confirmData, setConfirmData] = useState<ConfirmDraftDTO>({});
  const [issues, setIssues] = useState<string[]>([]);

  // Validar antes de confirmar
  const validateAndConfirm = async () => {
    try {
      const validation = await validateDraft(draft.id);

      if (!validation.valid) {
        setIssues(validation.issues);
        return;
      }

      onConfirm(confirmData);
    } catch (error) {
      setIssues([error.message]);
    }
  };

  if (!isOpen) return null;

  return (
    <Modal>
      <h2>Confirmar Lead - Pasar a ACTIVO</h2>

      {/* Mostrar problemas si los hay */}
      {issues.length > 0 && (
        <Alert type="warning">
          <h3>Problemas detectados:</h3>
          <ul>
            {issues.map(issue => <li key={issue}>{issue}</li>)}
          </ul>
        </Alert>
      )}

      {/* Mostrar datos del draft */}
      <div className="draft-summary">
        <p><strong>Nombre:</strong> {draft.datos.nombre}</p>
        <p><strong>Email:</strong> {draft.datos.email}</p>
        <p><strong>Empresa:</strong> {draft.datos.empresa}</p>
        <p><strong>Completitud:</strong> {draft.completitud}%</p>
      </div>

      {/* Campos adicionales para confirmar */}
      <select onChange={(e) => setConfirmData({
        ...confirmData,
        vendedorAsignado: e.target.value
      })}>
        <option>Seleccionar vendedor...</option>
        {/* Lista de vendedores */}
      </select>

      <input
        placeholder="Fuente del lead"
        onChange={(e) => setConfirmData({
          ...confirmData,
          fuente: e.target.value
        })}
      />

      <button onClick={validateAndConfirm}>Confirmar y Activar</button>
      <button onClick={onCancel}>Cancelar</button>
    </Modal>
  );
};
```

### 4. Hook useDraft
```typescript
// hooks/useDraft.ts

export const useDraft = () => {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<ApiError | null>(null);

  const getDrafts = async (filter?: DraftFilter) => {
    setLoading(true);
    try {
      const response = await apiClient.get('/drafts', { params: filter });
      return response.data;
    } catch (err) {
      setError(err);
      throw err;
    } finally {
      setLoading(false);
    }
  };

  const getDraft = async (draftId: string) => {
    setLoading(true);
    try {
      const response = await apiClient.get(`/drafts/${draftId}`);
      return response.data;
    } catch (err) {
      setError(err);
      throw err;
    } finally {
      setLoading(false);
    }
  };

  const saveDraft = async (data: CreateDraftLeadDTO) => {
    setLoading(true);
    try {
      const response = await apiClient.post('/drafts', data);
      return response.data;
    } catch (err) {
      setError(err);
      throw err;
    } finally {
      setLoading(false);
    }
  };

  const updateDraft = async (draftId: string, data: UpdateDraftLeadDTO) => {
    setLoading(true);
    try {
      const response = await apiClient.put(`/drafts/${draftId}`, data);
      return response.data;
    } catch (err) {
      setError(err);
      throw err;
    } finally {
      setLoading(false);
    }
  };

  const confirmDraft = async (draftId: string, data: ConfirmDraftDTO) => {
    setLoading(true);
    try {
      const response = await apiClient.post(`/drafts/${draftId}/confirm`, data);
      return response.data;
    } catch (err) {
      setError(err);
      throw err;
    } finally {
      setLoading(false);
    }
  };

  const deleteDraft = async (draftId: string) => {
    setLoading(true);
    try {
      await apiClient.delete(`/drafts/${draftId}`);
    } catch (err) {
      setError(err);
      throw err;
    } finally {
      setLoading(false);
    }
  };

  const validateDraft = async (draftId: string) => {
    setLoading(true);
    try {
      const response = await apiClient.post(`/drafts/${draftId}/validate`);
      return response.data;
    } catch (err) {
      setError(err);
      throw err;
    } finally {
      setLoading(false);
    }
  };

  return {
    loading,
    error,
    getDrafts,
    getDraft,
    saveDraft,
    updateDraft,
    confirmDraft,
    deleteDraft,
    validateDraft
  };
};
```

## 📊 Flujo Completo en UI

```
1. USER OPENS DRAFTS PAGE
   └─ <DraftLeadList /> loads
   └─ Hook useDraft() fetches list from API
   └─ Shows table with drafts

2. USER SEES INCOMPLETE DRAFT
   ├─ Name: ✅ Complete
   ├─ Email: ✅ Complete
   ├─ Phone: ❌ Missing
   ├─ Cargo: ❌ Missing
   └─ Completitud: 60%

3. USER CLICKS "EDITAR"
   └─ <DraftLeadForm draftId="..." /> opens
   └─ Loads existing data
   └─ User fills missing fields
   └─ Real-time validation on each field
   └─ Completitud updates to 100%

4. USER CLICKS "GUARDAR DRAFT"
   └─ API Call: PUT /api/drafts/{draftId}
   └─ Server validates and saves
   └─ Returns updated draft
   └─ UI confirms success

5. USER CLICKS "CONFIRMAR"
   └─ <DraftConfirmModal /> opens
   └─ Checks for issues
   └─ User assigns vendor
   └─ Selects source

6. USER CLICKS "CONFIRMAR Y ACTIVAR"
   └─ API Call: POST /api/drafts/{draftId}/confirm
   └─ Server validates all data
   └─ Converts to ACTIVO state
   └─ Returns updated lead
   └─ Redirect to leads page
```
